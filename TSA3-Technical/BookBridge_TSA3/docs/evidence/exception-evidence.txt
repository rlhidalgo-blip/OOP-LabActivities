public final class BookBridgeExceptionEvidence {
    private BookBridgeExceptionEvidence() { }
    public static void main(String[] args) {
        LibraryService service = new LibraryService(
                Clock.fixed(Instant.parse("2026-11-11T00:00:00Z"), ZoneOffset.UTC));
        service.registerMember(new StudentMember("S", "Student"));

    private static List<?> snapshot(LibraryService service) {
        return List.of(service.listMembers(), service.listBooks(), service.listActiveLoans(),
                service.listLoanHistory(), service.listPendingReservations(), service.getCheckedOutISBNs());
    }
    private static void capture(String label, Class<? extends RuntimeException> expected,
                                Runnable action, LibraryService service) {
        List<?> before = snapshot(service);
        try {
            action.run();
        } catch (RuntimeException actual) {
            if (!expected.isInstance(actual) || !before.equals(snapshot(service))) {
                throw new AssertionError("Wrong exception or changed collection state", actual);
            }
            System.out.println(label);
            System.out.println("  Actual: " + actual.getClass().getSimpleName() + ": " + actual.getMessage());
            System.out.println("  Expected type matched; all six collection snapshots unchanged: true");
            return;
        }
        throw new AssertionError("Expected rejection: " + label);
    }
